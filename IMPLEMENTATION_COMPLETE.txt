const service = initializeFeibotLiveTrackingService(config);

// 2. Start polling (automatically 2s interval)

// 3. Subscribe to updates

  console.log('Leaderboard updated:', data.leaderboard);
});

// 4. Query finish results
const results = await service.getFinishResult(eventUuid, {
  bib: '001,002,003',
});

// 5. Get all results
const allResults = await service.getAllResults(eventUuid);

// 6. Cleanup
service.destroy();
